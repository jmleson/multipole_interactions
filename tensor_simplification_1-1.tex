\section*{Interacting Multipoles of Rank 1 and 1:}
\begin{equation}
- 1 \cdot T_{\alpha{}\beta{}} \mu{}^{B}_{\alpha{}} \mu{}^{A}_{\beta{}} = R_{z}^{-5} \cdot -1 \cdot  \left[ \begin{array}{c} 
 		+ 3 \cdot R_{\alpha{}} \cdot R_{\beta{}} \cdot \mu{}^{B}_{\alpha{}} \cdot \mu{}^{A}_{\beta{}} 
 \\[0.2em] 		 - 1 \cdot R_{z}^{2} \cdot \delta{}\left(\alpha{}, \beta{}\right) \cdot \mu{}^{B}_{\alpha{}} \cdot \mu{}^{A}_{\beta{}}
 \end{array}\right] 
\end{equation}


\section*{Simplify deltas:}
\begin{subequations}\begin{gather}
- 1 \cdot T_{\alpha{}\beta{}} \mu{}^{B}_{\alpha{}} \mu{}^{A}_{\beta{}} = R_{z}^{-5} \cdot -1 \cdot  \left[ \begin{array}{c} 
 		+ 3 \cdot R_{\alpha{}} \cdot R_{\beta{}} \cdot \mu{}^{B}_{\alpha{}} \cdot \mu{}^{A}_{\beta{}} 
 \\[0.2em] 		 - 1 \cdot R_{z}^{2} \cdot \delta{}\left(\alpha{}, \alpha{}\right) \cdot \mu{}^{B}_{\alpha{}} \cdot \mu{}^{A}_{\alpha{}}
 \end{array}\right] 
\\ \xRightarrow{\text{cleaned up}}
 R_{z}^{-5} \cdot -1 \cdot  \left[ \begin{array}{c} 
 		+ 3 \cdot R_{\alpha{}} \cdot R_{\beta{}} \cdot \mu{}^{B}_{\alpha{}} \cdot \mu{}^{A}_{\beta{}} 
 \\[0.2em] 		 - 1 \cdot R_{z}^{2} \cdot \mu{}^{B}_{\alpha{}} \cdot \mu{}^{A}_{\alpha{}}
 \end{array}\right] 
\end{gather}\end{subequations}


\section*{Simplify R:}
\begin{subequations}\begin{gather}
- 1 \cdot T_{\alpha{}\beta{}} \mu{}^{B}_{\alpha{}} \mu{}^{A}_{\beta{}} = R_{z}^{-5} \cdot -1 \cdot  \left[ \begin{array}{c} 
 		+ 3 \cdot R_{z} \cdot R_{z} \cdot \mu{}^{B}_{z} \cdot \mu{}^{A}_{z} 
 \\[0.2em] 		 - 1 \cdot R_{z}^{2} \cdot \mu{}^{B}_{\alpha{}} \cdot \mu{}^{A}_{\alpha{}}
 \end{array}\right] 
\\ \xRightarrow{\text{cleaned up}}
 R_{z}^{-5} \cdot -1 \cdot  \left[ \begin{array}{c} 
 		- 1 \cdot R_{z}^{2} \cdot \mu{}^{B}_{\alpha{}} \cdot \mu{}^{A}_{\alpha{}}
 \end{array}\right] 
\end{gather}\end{subequations}


\section*{Simplify Dipoles:}
\begin{equation}
- 1 \cdot T_{\alpha{}\beta{}} \mu{}^{B}_{\alpha{}} \mu{}^{A}_{\beta{}} = R_{z}^{-5} \cdot -1 \cdot  \left[ \begin{array}{c} 
 		- 1 \cdot R_{z}^{2} \cdot \mu{}^{B}_{y} \cdot \mu{}^{A}_{y}
 \end{array}\right] 
\end{equation}


\section*{Bombining everything:}

 		+ 1 \cdot R_{z}^{-3} \cdot \mu{}^{B}_{y} \cdot \mu{}^{A}_{y}

\end{equation}
